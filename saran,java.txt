import java.util.Scanner;

public class DrivingLicenceEligibility {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        System.out.print("Enter age: ");
        int age = input.nextInt();

        // Relational operator: age >= 18
        // Ternary operator: condition ? trueValue : falseValue
        String status = (age >= 18) ? "Eligible" : "Not Eligible";

        System.out.println(status);

        input.close();
    }
}

OUTPUT:

Enter age: 18
Eligible

7.LOGIN ACCOUNT CONTROLLER

import java.util.Scanner;

public class LoginAttemptController {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the maximum number of failed attempts: ");
        int maxAttempts = scanner.nextInt();

        System.out.print("Enter the current number of failed attempts: ");
        int failedAttempts = scanner.nextInt();

        if (maxAttempts <= 0 || failedAttempts < 0) {
            System.out.println("Please enter a positive maximum and a non-negative failed-attempt count.");
        } else if (failedAttempts < maxAttempts) {
            System.out.print("Did the login attempt fail? (true/false): ");
            boolean loginFailed = scanner.nextBoolean();

            if (loginFailed) {
                failedAttempts += 1;
            } else {
                System.out.println("Login successful.");
            }

            if (failedAttempts < maxAttempts && maxAttempts > 0) {
                System.out.println("Next login attempt is allowed.");
            } else {
                System.out.println("Account is locked.");
            }
        } else {
            System.out.println("Account is locked.");
        }

        scanner.close();
    }
}


OUTPUT

Enter the maximum number of failed attempts: 6 
Enter the current number of failed attempts: 4
Did the login attempt fail? (true/false): 
TRUE
Next login attempt is allowed.

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter battery percentage: ");
        int batteryLevel = scanner.nextInt();

        System.out.print("Enter temperature in °C: ");
        double temperature = scanner.nextDouble();

        boolean chargingAllowed =
                batteryLevel < 90 && temperature >= 10 && temperature <= 45;

        if (chargingAllowed) {

        String status = chargingAllowed
                ? "Charging allowed. Battery level is now " + batteryLevel + "%."
                : "Charging not allowed.";

        System.out.println(status);
        scanner.close();
    }
}

12.Ride Fare Calculator

import java.util.Scanner;

public class RideFareCalculator {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter distance in kilometres: ");
        double distance = scanner.nextDouble();

        System.out.print("Enter waiting time in minutes: ");
        int waitingMinutes = scanner.nextInt();

        System.out.print("Is the customer a premium member? (yes/no): ");
        boolean premiumMember = scanner.next().equalsIgnoreCase("yes");

        if (distance <= 0 || waitingMinutes < 0) {
            System.out.println("Invalid distance or waiting time.");
        } else {
            double fare = 50 + distance * 12;
            fare += Math.max(0, waitingMinutes - 5) * 2;

            boolean discountApplies = premiumMember && fare > 300;
            double finalFare = discountApplies ? fare * 0.90 : fare;

            System.out.printf("Final ride fare: ₹%.2f%n", finalFare);
        }

        scanner.close();
    }
}


Output:

Enter distance in kilometres: 76
Enter waiting time in minutes: 45
Is the customer a premium member? (yes/no): no
Final ride fare: ?1042.00

14.Secure Transaction Validator


import java.util.Scanner;

public class SecureTransactionValidator {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter account balance: ₹");
        double balance = scanner.nextDouble();

        System.out.print("Enter transfer amount: ₹");
        double transferAmount = scanner.nextDouble();

        System.out.print("Enter daily transfer limit: ₹");
        double dailyLimit = scanner.nextDouble();

        System.out.print("Is the account active? (true/false): ");
        boolean accountActive = scanner.nextBoolean();

        if (transferAmount > 0
                && transferAmount <= balance
                && transferAmount <= dailyLimit
                && accountActive) {
            balance -= transferAmount;
            System.out.println("Transaction successful.");
        } else {
            System.out.println("Transaction declined.");
        }

        System.out.printf("Remaining balance: ₹%.2f%n", balance);
        scanner.close();
    }
}

Enter transfer amount: ?450000
Enter daily transfer limit: ?100000

import java.util.Scanner;

public class SmartExamRegistrationSystem {
    public static void main(String[] args) {

        System.out.print("Enter attendance percentage: ");
        double attendance = scanner.nextDouble();

        System.out.print("Is there an approved medical certificate? (true/false): ");
        boolean medicalCertificateApproved = scanner.nextBoolean();

        System.out.print("Has the examination fee been paid? (true/false): ");
        boolean feePaid = scanner.nextBoolean();

        System.out.print("Have all assignments been submitted? (true/false): ");
        boolean assignmentsSubmitted = scanner.nextBoolean();

        boolean attendanceEligible = attendance >= 75
                || (attendance >= 65 && medicalCertificateApproved);

        String result = (attendanceEligible && feePaid && assignmentsSubmitted)
                ? "Registration Successful"
                : "Registration Rejected";

        System.out.println(result);
        scanner.close();
    }
}

Is there an approved medical certificate? (true/false): true
Has the examination fee been paid? (true/false): false
